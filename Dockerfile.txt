# ---- 1. Base image ----
FROM ubuntu:24.04

# ---- 2. Install sudo (and any other packages you need) ----
RUN apt-get update && \
    apt-get install -y --no-install-recommends sudo && \
    rm -rf /var/lib/apt/lists/*

# ---- 3. Create the non‑root user & add to the sudo group ----
RUN groupadd -r username && \
    useradd -m -r -g username -s /bin/bash username && \
    usermod -aG sudo username

# ---- 4. Allow 'username' NOPASSWD sudo (optional but handy for CI) ----
RUN echo "username ALL=(ALL) NOPASSWD: ALL" >> /etc/sudoers.d/username && \
    chmod 0440 /etc/sudoers.d/username

# ---- 5. Working directory ----
WORKDIR /home/username

# ---- 6. Switch to the non‑root user ----
USER username

# ---- 7. Default command (replace if you need something else) ----
CMD ["bash"]
